// @name         Zoom Transcriber Pro
// @namespace    http://tampermonkey.net/

(function () {
    'use strict';

    // ============================================================
    // CONFIG
    // ============================================================

    const AUTO_SAVE_INTERVAL   = 3 * 60 * 1000;   // 3 minutes
    const KEYWORD_COOLDOWN     = 2 * 60 * 1000;   // 2 minutes silence after alert
    const FINALIZE_PAUSE       = 3000;             // ms of no update → line is "done"

    const KEYWORDS = [
        "add", "your", "keywords", "here"
    ];

    // ============================================================
    // STATE
    // ============================================================

    let currentMode        = null;   // "autosave" | "notes"

    // The finalized transcript (array of strings like "[HH:MM:SS] texto")
    let transcript         = [];

    // The line currently being built by Zoom (not yet finalized)
    let activeLine         = "";     // raw text
    let activeLineTS       = "";     // timestamp when active line started
    let activeLineTimer    = null;   // setTimeout handle — fires when line goes quiet

    // DOM panel (notes mode)
    let notesPanel         = null;
    let notesBody          = null;
    let activeLineEl       = null;   // the <div> in the panel for the in-progress line

    // Keyword alert cooldown
    const recentlyAlerted  = new Map();

    // Flags
    let observerStarted    = false;
    let controlsStarted    = false;
    let saveTimerStarted   = false;
    let autoSaveFilename   = "";

    // ============================================================
    // UTILITIES
    // ============================================================

    function norm(text) {
        return (text || "").replace(/\u00a0/g, " ").replace(/\s+/g, " ").trim();
    }

    function ts() {
        return new Date().toLocaleTimeString("es-HN", {
            hour: "2-digit", minute: "2-digit", second: "2-digit"
        });
    }

    function todayFilename() {
        const d  = new Date();
        const dd = String(d.getDate()).padStart(2, "0");
        const mm = String(d.getMonth() + 1).padStart(2, "0");
        const yy = d.getFullYear();
        return `CONTROL-TRANSCRIPCION-${dd}-${mm}-${yy}.txt`;
    }

    function log(...a) {
        console.log("%c[Zoom Transcriber]", "color:#0e71eb;font-weight:bold;", ...a);
    }

    // ============================================================
    // CORE CAPTION ENGINE
    //
    // Zoom keeps updating the SAME DOM node while a speaker talks.
    // The text grows word by word, e.g.:
    //   "Hola"  →  "Hola a todos"  →  "Hola a todos, bienvenidos"
    // Then a NEW node appears for the next sentence.
    //
    // Strategy:
    //   • Keep one "active line" that we update as Zoom updates it.
    //   • When the text stops changing for FINALIZE_PAUSE ms, lock it
    //     in as a finalized transcript entry.
    //   • New incoming text that STARTS with words not in the active
    //     line means a brand-new sentence → finalize current, start new.
    // ============================================================

    function isExtensionOf(existing, incoming) {
        if (!existing) return true;
        const e = existing.toLowerCase().trim();
        const i = incoming.toLowerCase().trim();
        // Incoming starts with existing
        if (i.startsWith(e)) return true;
        // Incoming contains the last 30 chars of existing (handles word-wrap resets)
        const tail = e.slice(-Math.min(30, e.length));
        if (tail.length > 8 && i.includes(tail)) return true;
        // Existing starts with incoming (Zoom showed a shorter version momentarily)
        if (e.startsWith(i)) return true;
        // Large word overlap: if >60% of incoming words are in existing, it's the same line
        const eWords = e.split(" ").filter(Boolean);
        const iWords = i.split(" ").filter(Boolean);
        if (iWords.length < 3) return false;
        const overlap = iWords.filter(w => eWords.includes(w)).length;
        return overlap / iWords.length > 0.6;
    }

    function overlapMerge(existing, incoming) {
        // Find the longest suffix of `existing` that is a prefix of `incoming`,
        // then append the remainder of `incoming`.
        const eWords = existing.split(" ").filter(Boolean);
        const iWords = incoming.split(" ").filter(Boolean);
        const max    = Math.min(eWords.length, iWords.length);

        for (let size = max; size >= 1; size--) {
            const eSuffix = eWords.slice(-size).join(" ").toLowerCase();
            const iPrefix = iWords.slice(0, size).join(" ").toLowerCase();
            if (eSuffix === iPrefix) {
                const rest = iWords.slice(size).join(" ");
                return rest ? existing + " " + rest : existing;
            }
        }
        return null; // no overlap found
    }

    function processCaption(rawText) {
        const text = norm(rawText);
        if (!text || text.length < 3)  return;
        if (isProbablyUI(text))        return;
        if (text === activeLine)       return;  // identical — nothing changed

        checkKeywords(text);

        // ── Case 1: active line is empty → start fresh ──────────
        if (!activeLine) {
            activeLine   = text;
            activeLineTS = ts();
            scheduleFinalize();
            onActiveLineChanged(true);
            return;
        }

        // ── Case 2: incoming is longer and extends the active line ──
        if (text.length > activeLine.length && isExtensionOf(activeLine, text)) {
            activeLine = text;
            scheduleFinalize();
            onActiveLineChanged(false);
            return;
        }

        // ── Case 3: incoming overlaps end of active line ─────────
        const merged = overlapMerge(activeLine, text);
        if (merged && merged !== activeLine) {
            activeLine = merged;
            scheduleFinalize();
            onActiveLineChanged(false);
            return;
        }

        // ── Case 4: incoming is a subset (Zoom rolled back) ──────
        if (activeLine.toLowerCase().includes(text.toLowerCase())) {
            return; // already have more, ignore
        }

        // ── Case 4b: try overlap merge one more time before giving up ──
        const merged2 = overlapMerge(activeLine, text);
        if (merged2 && merged2 !== activeLine) {
            activeLine = merged2;
            scheduleFinalize();
            onActiveLineChanged(false);
            return;
        }

        // ── Case 5: genuinely new sentence ───────────────────────
        finalizeLine();
        activeLine   = text;
        activeLineTS = ts();
        scheduleFinalize();
        onActiveLineChanged(true);
    }

    // Reset the finalize timer every time the line changes
    function scheduleFinalize() {
        if (activeLineTimer) clearTimeout(activeLineTimer);
        activeLineTimer = setTimeout(finalizeLine, FINALIZE_PAUSE);
    }

    // Lock the active line into the permanent transcript
    function finalizeLine() {
        if (activeLineTimer) { clearTimeout(activeLineTimer); activeLineTimer = null; }
        if (!activeLine) return;

        const entry = `[${activeLineTS}] ${activeLine}`;
        transcript.push(entry);
        log("Finalized:", entry);

        onLineFinalized(entry);

        activeLine   = "";
        activeLineTS = "";
        activeLineEl = null;
    }

    // ============================================================
    // MODE CALLBACKS
    // ============================================================

    // Called when the in-progress line changes
    function onActiveLineChanged(isNew) {
        // Push to notes panel whenever it exists, regardless of mode
        if (notesBody) updateNotesActive(isNew);
    }

    // Called when a line is locked in
    function onLineFinalized(entry) {
        // Push to notes panel whenever it exists, regardless of mode
        if (notesBody) finalizeNotesLine(entry);
    }

    // ============================================================
    // UI HELPERS
    // ============================================================

    function isProbablyUI(text) {
        if (text.length > 120) return false;
        const lower = text.toLowerCase();
        // Exact matches (short UI labels)
        const exact = [
            "mute", "unmute", "start video", "stop video", "share screen",
            "participants", "chat", "settings", "leave meeting", "end meeting",
            "more", "reactions",
            // Spanish Zoom UI
            "vista previa", "preview",
            "ocultar subtítulos", "ocultar subtitulos",
            "mostrar subtítulos", "mostrar subtitulos",
            "los subtítulos están activados",
            "los subtitulos estan activados",
            "subtítulos activados", "subtitulos activados",
            "cancelar silenciar", "silenciar", "reactivar audio",
            "compartir", "salir", "grabar", "más", "reaccionar",
            "ver", "cerrar", "zoom ai"
        ];
        if (exact.some(x => lower === x || lower === x + ".")) return true;
        // Starts-with matches for notification banners
        const startsWith = [
            "los subtítulos", "los subtitulos",
            "subtitles are", "captions are"
        ];
        if (startsWith.some(x => lower.startsWith(x))) return true;
        return false;
    }

    // ============================================================
    // KEYWORD ALERTS
    // ============================================================

    function checkKeywords(text) {
        if (!KEYWORDS.length) return;
        const lower = text.toLowerCase();
        for (const kw of KEYWORDS) {
            if (!lower.includes(kw.toLowerCase())) continue;
            const now  = Date.now();
            const prev = recentlyAlerted.get(kw) || 0;
            if (now - prev < KEYWORD_COOLDOWN) continue;
            recentlyAlerted.set(kw, now);
            triggerAlert(kw, text);
        }
    }

    function triggerAlert(kw, fullText) {
        log("KEYWORD:", kw);

        // Audio — three short beeps
        try {
            const Ctx = window.AudioContext || window.webkitAudioContext;
            if (Ctx) {
                const ctx = new Ctx();
                [0, 0.3, 0.6].forEach(off => {
                    const osc  = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type             = "sine";
                    osc.frequency.value  = 880;
                    gain.gain.value      = 0.5;
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start(ctx.currentTime + off);
                    osc.stop(ctx.currentTime + off + 0.2);
                });
            }
        } catch (e) {}

        // Desktop notification
        const msg = fullText.length > 100 ? fullText.slice(0, 97) + "…" : fullText;
        function sendNotif() {
            try {
                if (typeof GM_notification !== "undefined") {
                    GM_notification({
                        title: `🔔 Zoom: "${kw}" mencionado`,
                        text:  msg,
                        timeout: 8000
                    });
                } else {
                    new Notification(`🔔 Zoom: "${kw}" mencionado`, {
                        body: msg,
                        icon: "https://st1.zoom.us/zoom.ico"
                    });
                }
            } catch (e) {}
        }

        if (typeof GM_notification !== "undefined") {
            sendNotif();
        } else if ("Notification" in window) {
            if (Notification.permission === "granted") sendNotif();
            else if (Notification.permission !== "denied")
                Notification.requestPermission().then(p => { if (p === "granted") sendNotif(); });
        }

        // Red banner
        showBanner(`🔔 "${kw}" mencionado!`);
    }

    function showBanner(msg) {
        let b = document.getElementById("tm-kw-banner");
        if (!b) {
            b = document.createElement("div");
            b.id = "tm-kw-banner";
            Object.assign(b.style, {
                position:     "fixed",
                top:          "60px",
                left:         "50%",
                transform:    "translateX(-50%)",
                zIndex:       "2147483647",
                padding:      "12px 28px",
                borderRadius: "10px",
                background:   "#e74c3c",
                color:        "white",
                fontWeight:   "bold",
                fontSize:     "16px",
                boxShadow:    "0 4px 20px rgba(0,0,0,.5)",
                transition:   "opacity 0.6s",
                pointerEvents:"none"
            });
            document.body.appendChild(b);
        }
        b.textContent  = msg;
        b.style.opacity = "1";
        clearTimeout(b._timer);
        b._timer = setTimeout(() => { b.style.opacity = "0"; }, 5000);
    }

    // ============================================================
    // CAPTION DOM SCANNING
    // ============================================================

    function isVisible(el) {
        if (!el) return false;
        const s = window.getComputedStyle(el);
        if (s.display === "none" || s.visibility === "hidden" || s.opacity === "0") return false;
        const r = el.getBoundingClientRect();
        return r.width > 0 && r.height > 0;
    }

    function scanCaptions() {
        const selectors = [
            // Zoom Web exact classes (from live DOM inspection)
            '.live-transcription-subtitle__item',
            '.live-transcription-subtitle__content',
            '.live-transcription-subtitle__box',
            // Generic fallbacks
            '[class*="caption"]', '[class*="Caption"]',
            '[class*="subtitle"]', '[class*="Subtitle"]',
            '[class*="transcript"]', '[class*="Transcript"]',
            '[aria-label*="caption" i]', '[aria-label*="subtitle" i]',
            '[aria-label*="transcript" i]',
            '[data-testid*="caption" i]', '[data-testid*="subtitle" i]'
        ];
        const seen = new Set();
        for (const sel of selectors) {
            try {
                document.querySelectorAll(sel).forEach(el => {
                    if (seen.has(el) || !isVisible(el)) return;
                    seen.add(el);
                    const text = norm(el.innerText || el.textContent || "");
                    if (text && text.length <= 600) processCaption(text);
                });
            } catch (_) {}
        }
        // Also scan inside any iframes on the same origin
        try {
            document.querySelectorAll("iframe").forEach(iframe => {
                try {
                    const idoc = iframe.contentDocument;
                    if (!idoc) return;
                    ['.live-transcription-subtitle__item',
                     '[class*="caption"]','[class*="subtitle"]'].forEach(sel => {
                        idoc.querySelectorAll(sel).forEach(el => {
                            if (seen.has(el)) return;
                            seen.add(el);
                            const text = norm(el.innerText || el.textContent || "");
                            if (text && text.length <= 600) processCaption(text);
                        });
                    });
                } catch (_) {}
            });
        } catch (_) {}
    }

    function startCaptionObserver() {
        if (observerStarted) return;
        if (!document.body) { setTimeout(startCaptionObserver, 500); return; }
        observerStarted = true;

        // Throttle: scan at most once every 200ms no matter how many mutations fire
        let scanPending = false;
        const onMutation = () => {
            if (scanPending) return;
            scanPending = true;
            setTimeout(() => { scanCaptions(); scanPending = false; }, 200);
        };

        // Observe the top-level document
        new MutationObserver(onMutation)
            .observe(document.body, { childList: true, subtree: true, characterData: true });

        // Also observe inside iframes once they load (Zoom renders subtitles in an iframe)
        function observeIframes() {
            document.querySelectorAll("iframe").forEach(iframe => {
                if (iframe._tmObserved) return;
                try {
                    const idoc = iframe.contentDocument;
                    if (!idoc || !idoc.body) return;
                    iframe._tmObserved = true;
                    new MutationObserver(onMutation)
                        .observe(idoc.body, { childList: true, subtree: true, characterData: true });
                    log("Observing iframe:", iframe.id || iframe.src?.slice(0, 60));
                } catch (_) {}
            });
        }

        // Watch for new iframes being added (throttled to avoid loop)
        let iframeCheckPending = false;
        new MutationObserver(() => {
            if (iframeCheckPending) return;
            iframeCheckPending = true;
            setTimeout(() => { observeIframes(); iframeCheckPending = false; }, 500);
        }).observe(document.body, { childList: true });

        observeIframes();
        scanCaptions();
        log("Caption observer started.");
    }

    // ============================================================
    // AUTO-ENABLE ZOOM CAPTIONS
    // ============================================================

    function findCaptionControl() {
        // Direct class match (from live inspection of your Zoom page)
        const byClass = document.querySelector("button.new-lt-button");
        if (byClass && isVisible(byClass)) return byClass;

        const positive = [
            "mostrar subtítulos", "mostrar subtitulos",
            "activar subtítulos", "subtítulos", "subtitulos",
            "show captions", "show subtitles", "closed captions",
            "captions", "subtitles", "live transcript", "transcription"
        ];
        const negative = [
            "ocultar subtítulos", "ocultar subtitulos",
            "hide captions", "stop transcription", "disable captions",
            "more options for captions", "más opciones de subtítulos"
        ];

        for (const el of document.querySelectorAll('button, [role="button"], [role="menuitem"], a')) {
            if (!isVisible(el)) continue;
            const t = (el.innerText || el.textContent || el.getAttribute("aria-label") || "")
                .replace(/\s+/g, " ").trim().toLowerCase();
            if (!t) continue;
            if (negative.some(w => t.includes(w))) continue;
            if (positive.some(w => t.includes(w))) return el;
        }
        return null;
    }

    function autoEnableCaptions() {
        if (controlsStarted) return;
        const btn = findCaptionControl();
        if (!btn) return;
        controlsStarted = true;
        log("Clicking:", btn.innerText || btn.getAttribute("aria-label"));
        btn.click();

        // If a submenu appeared, click the right item inside it
        setTimeout(() => {
            const menuKw = [
                "mostrar subtítulos", "mostrar subtitulos",
                "show subtitles", "show captions",
                "habilitar", "enable",
                "ver transcripción", "view full transcript"
            ];
            let clicked = false;
            for (const el of document.querySelectorAll(
                '[role="menuitem"], [role="option"], [class*="menu-item"], li'
            )) {
                if (!isVisible(el)) continue;
                const t = (el.innerText || el.textContent || "").replace(/\s+/g, " ").trim().toLowerCase();
                if (menuKw.some(k => t.includes(k))) {
                    log("Submenu:", t);
                    el.click();
                    clicked = true;
                    break;
                }
            }
            if (!clicked) {
                const retry = findCaptionControl();
                if (retry && retry !== btn) retry.click();
            }
            startCaptionObserver();
        }, 800);
    }

    // ============================================================
    // MODE: AUTO-SAVE
    // One file, rebuilt in full each save cycle
    // ============================================================

    function startAutoSaveMode() {
        currentMode      = "autosave";
        autoSaveFilename = todayFilename();
        log("Auto-save mode →", autoSaveFilename);
        if (saveTimerStarted) return;
        saveTimerStarted = true;
        setInterval(() => { if (transcript.length || activeLine) saveNow(); }, AUTO_SAVE_INTERVAL);
        log("Auto-save every 3 min.");
    }

    function saveNow() {
        const lines = [...transcript];
        if (activeLine) lines.push(`[${activeLineTS}] ${activeLine} …`);
        if (!lines.length) return;
        const blob = new Blob([lines.join("\n")], { type: "text/plain;charset=utf-8" });
        const url  = URL.createObjectURL(blob);
        const a    = Object.assign(document.createElement("a"), {
            href: url, download: autoSaveFilename || todayFilename()
        });
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(() => URL.revokeObjectURL(url), 1000);
        log("Saved:", autoSaveFilename);
    }

    // Add any line to the stream
    function appendStreamLine(text, type) {
        // Re-grab notesBody from DOM in case the reference was lost
        if (!notesBody) notesBody = document.getElementById("tm-notes-body");
        if (!notesBody) return null;
        const styles = {
            transcript: "color:#9ca3af; border-bottom:1px solid #1f2937;",
            active:     "color:#60a5fa; font-style:italic; border-bottom:1px solid #1e3a5f;",
            note:       "color:#fcd34d; border-left:3px solid #f59e0b; padding-left:8px; margin:4px 0; background:rgba(245,158,11,0.07); border-radius:0 4px 4px 0;"
        };
        const d = el("div", {
            "data-line": type,
            style: `padding:2px 0 4px; margin-bottom:2px; ${styles[type] || styles.transcript}`,
            textContent: text
        });
        notesBody.appendChild(d);
        scrollBottom();
        return d;
    }

    function insertNote(input) {
        const text = input.value.trim();
        if (!text) return;
        appendStreamLine(text, "note");
        input.value = "";
        input.style.height = "auto";
        input.focus();
    }

    function updateNotesActive(isNew) {
        if (!notesBody) notesBody = document.getElementById("tm-notes-body");
        if (!notesBody) return;
        const text = `[${activeLineTS}] ${activeLine}`;
        // Only create a new element if we truly don't have one in the DOM
        const elExists = activeLineEl && activeLineEl.isConnected;
        if (!elExists) {
            activeLineEl = appendStreamLine(text, "active");
        } else {
            // Update in place — no new element, no new timestamp flicker
            activeLineEl.textContent = text;
            scrollBottom();
        }
    }

    function finalizeNotesLine(entry) {
        if (!notesBody) notesBody = document.getElementById("tm-notes-body");
        if (!notesBody) return;
        if (activeLineEl && activeLineEl.parentNode) {
            activeLineEl.style.color      = "#9ca3af";
            activeLineEl.style.fontStyle  = "";
            activeLineEl.style.borderBottom = "1px solid #1f2937";
            activeLineEl.setAttribute("data-line", "transcript");
        } else {
            appendStreamLine(entry, "transcript");
        }
        activeLineEl = null;
    }

    function scrollBottom() {
        if (!notesBody) notesBody = document.getElementById("tm-notes-body");
        if (!notesBody) return;
        const atBot = notesBody.scrollHeight - notesBody.scrollTop - notesBody.clientHeight < 120;
        if (atBot) notesBody.scrollTop = notesBody.scrollHeight;
    }

    // ============================================================
    // DOM HELPERS
    // ============================================================

    function el(tag, props = {}) {
        const e = document.createElement(tag);
        for (const [k, v] of Object.entries(props)) {
            if (k === "style")    e.style.cssText = v;
            else if (k === "innerHTML")  e.innerHTML = v;
            else if (k === "textContent") e.textContent = v;
            else if (k === "contentEditable") e.contentEditable = v;
            else if (k === "spellcheck") e.spellcheck = v;
            else e.setAttribute(k, v);
        }
        return e;
    }

    function mkBtn(label, bg, onClick) {
        const b = el("button", {
            textContent: label,
            style: `
                padding:6px 12px; border:none; border-radius:7px;
                background:${bg}; color:#fff; font-size:12px;
                font-weight:bold; cursor:pointer;
            `
        });
        b.addEventListener("click", onClick);
        return b;
    }

    function makeDraggable(panel, handle) {
        let ox = 0, oy = 0, mx = 0, my = 0;
        handle.addEventListener("mousedown", e => {
            e.preventDefault();
            ox = panel.offsetLeft || (window.innerWidth - panel.offsetWidth - 20);
            oy = panel.offsetTop  || 60;
            mx = e.clientX; my = e.clientY;
            document.addEventListener("mousemove", drag);
            document.addEventListener("mouseup",   stop);
        });
        function drag(e) {
            panel.style.right = "auto";
            panel.style.left  = (ox + e.clientX - mx) + "px";
            panel.style.top   = (oy + e.clientY - my) + "px";
        }
        function stop() {
            document.removeEventListener("mousemove", drag);
            document.removeEventListener("mouseup",   stop);
        }
    }

    // ============================================================
    // SCANNER LOOP
    // ============================================================

    function startControlScanner() {
        const timer = setInterval(() => {
            if (!controlsStarted) autoEnableCaptions();
            if (document.body && !document.getElementById("tm-zoom-controls")) createControlBar();
            // No scanCaptions here — the MutationObserver handles it
        }, 1500);

        const captionRetry = setInterval(() => {
            if (controlsStarted) { clearInterval(captionRetry); return; }
            autoEnableCaptions();
        }, 3000);

        setTimeout(() => { clearInterval(timer); clearInterval(captionRetry); }, 300000);
    }

    // ============================================================
    // FINAL SAVE ON UNLOAD
    // ============================================================

    window.addEventListener("beforeunload", () => {
        if (transcript.length || activeLine) saveNow();
    });

    // ============================================================
    // BOOT
    // ============================================================

    function start() {
        log("🚀 Zoom Transcriber Pro v6.0");
        if ("Notification" in window && Notification.permission === "default")
            Notification.requestPermission();
        startControlScanner();
        startCaptionObserver();
    }

    if (document.readyState === "loading")
        document.addEventListener("DOMContentLoaded", start, { once: true });
    else
        start();
